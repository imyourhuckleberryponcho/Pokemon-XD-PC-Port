file(GLOB _aurora_patches CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/third_party/patches/aurora/*.patch")
list(SORT _aurora_patches)
set(_aurora_dir "${PROJECT_SOURCE_DIR}/third_party/aurora")
set(_aurora_stamp_file "${_aurora_dir}/.port_patches_applied")
set(_aurora_stamp "")
foreach (_patch ${_aurora_patches})
  file(SHA1 "${_patch}" _hash)
  get_filename_component(_name "${_patch}" NAME)
  string(APPEND _aurora_stamp "${_name} ${_hash}\n")
endforeach ()
set(_aurora_old_stamp "")
if (EXISTS "${_aurora_stamp_file}")
  file(READ "${_aurora_stamp_file}" _aurora_old_stamp)
endif ()
if (NOT _aurora_old_stamp STREQUAL _aurora_stamp)
  message(STATUS "Aurora patches changed: resetting the Aurora checkout and applying them all")
  execute_process(COMMAND git checkout -- .
    WORKING_DIRECTORY "${_aurora_dir}" RESULT_VARIABLE _result)
  if (NOT _result EQUAL 0)
    message(FATAL_ERROR "Couldn't reset the Aurora checkout (${_aurora_dir})")
  endif ()
  execute_process(COMMAND git clean -fdq -e ".*"
    WORKING_DIRECTORY "${_aurora_dir}" RESULT_VARIABLE _result)
  if (NOT _result EQUAL 0)
    message(FATAL_ERROR "Couldn't remove the patches' added files in ${_aurora_dir}")
  endif ()
  file(REMOVE "${_aurora_stamp_file}")
  foreach (_patch ${_aurora_patches})
    execute_process(COMMAND git apply "${_patch}"
      WORKING_DIRECTORY "${_aurora_dir}" RESULT_VARIABLE _result)
    if (NOT _result EQUAL 0)
      message(FATAL_ERROR "Aurora patch doesn't apply: ${_patch}")
    endif ()
    message(STATUS "Applied Aurora patch ${_patch}")
  endforeach ()
  file(WRITE "${_aurora_stamp_file}" "${_aurora_stamp}")
endif ()

if (MSVC AND DEFINED ENV{XD_TARGET_NOOPT} AND NOT "$ENV{XD_TARGET_NOOPT}" STREQUAL "")
  function(_xd_collect_targets _dir _out)
    get_property(_t DIRECTORY "${_dir}" PROPERTY BUILDSYSTEM_TARGETS)
    get_property(_subs DIRECTORY "${_dir}" PROPERTY SUBDIRECTORIES)
    foreach (_sub ${_subs})
      _xd_collect_targets("${_sub}" _more)
      list(APPEND _t ${_more})
    endforeach ()
    set(${_out} ${_t} PARENT_SCOPE)
  endfunction()
  _xd_collect_targets("${PROJECT_SOURCE_DIR}" _all_targets)
  set(_noopt_targets "")
  foreach (_t ${_all_targets})
    get_target_property(_type ${_t} TYPE)
    if (_type MATCHES "^(STATIC_LIBRARY|SHARED_LIBRARY|MODULE_LIBRARY|OBJECT_LIBRARY|EXECUTABLE)$")
      set(_skip FALSE)
      foreach (_re $ENV{XD_TARGET_NOOPT_EXCEPT}) # (e.g. xd_game: keep the game optimised)
        if (_t MATCHES "${_re}")
          set(_skip TRUE)
        endif ()
      endforeach ()
      if (_skip)
        continue()
      endif ()
      foreach (_re $ENV{XD_TARGET_NOOPT})
        if (_t MATCHES "${_re}")
          target_compile_options(${_t} PRIVATE $<$<COMPILE_LANGUAGE:C,CXX>:/Od /Ob0>)
          list(APPEND _noopt_targets ${_t})
          break()
        endif ()
      endforeach ()
    endif ()
  endforeach ()
  message(STATUS "xd: XD_TARGET_NOOPT: ${_noopt_targets}")
endif ()
