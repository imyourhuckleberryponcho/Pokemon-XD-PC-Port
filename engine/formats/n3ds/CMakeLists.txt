target_compile_features(engine_n3ds PUBLIC cxx_std_17)
find_package(Threads REQUIRED)
target_link_libraries(engine_n3ds PUBLIC Threads::Threads engine_pbr)
if (TARGET platform_disc) # the XD disc is read through the port's disc library (nod): ISO / GCM / RVZ / WIA / GCZ / CISO, as the game reads it
  target_link_libraries(engine_n3ds PUBLIC platform_disc)
  target_compile_definitions(engine_n3ds PRIVATE XD_N3DS_PORT_DISC=1)
endif ()

  target_compile_options(engine_n3ds PRIVATE -ffp-contract=off)
else ()
  target_compile_definitions(engine_n3ds PRIVATE _CRT_SECURE_NO_WARNINGS)
endif ()
